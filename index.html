const express = require("express"); const helmet = require("helmet"); const cookieParser = require("cookie-parser"); const bcrypt = require("bcryptjs"); const jwt = require("jsonwebtoken"); const Database = require("better-sqlite3"); const multer = require("multer"); const crypto = require("crypto"); const fs = require("fs"); const path = require("path"); const app = express(); const PORT = Number(process.env.PORT || 3000); const JWT_SECRET = process.env.JWT_SECRET || "SUPER_SECRET_CHANGE_ME"; const db = new Database("radcare.db"); // Middleware app.use(helmet({ contentSecurityPolicy: false })); app.use(express.json({ limit: "2mb" })); app.use(cookieParser()); app.use(express.static(path.join(__dirname, "public"))); const uploadDir = path.join(__dirname, "private_uploads"); if (!fs.existsSync(uploadDir)) fs.mkdirSync(uploadDir); // Helpers const now = () => new Date().toISOString(); const id = () => crypto.randomUUID(); const publicUser = (u) => ({ id: u.id, name: u.name, email: u.email, role: u.role, status: u.status }); // Database Initialization db.exec(` CREATE TABLE IF NOT EXISTS users (id TEXT PRIMARY KEY, name TEXT, email TEXT UNIQUE, password_hash TEXT, country TEXT, role TEXT, status TEXT, created_at TEXT); CREATE TABLE IF NOT EXISTS patient_profiles (id TEXT PRIMARY KEY, user_id TEXT UNIQUE, phone TEXT, date_of_birth TEXT, emergency_contact TEXT, created_at TEXT); -- Add other tables here... `); // Auth Middleware function auth(req, res, next) { const token = req.cookies.radcare_token; if (!token) return res.status(401).json({ error: "Unauthorized" }); try { const decoded = jwt.verify(token, JWT_SECRET); req.user = db.prepare("SELECT * FROM users WHERE id=?").get(decoded.sub); next(); } catch { res.status(401).json({ error: "Invalid token" }); } } // --- ROUTES --- app.post("/api/auth/register", async (req, res) => { const { name, email, password, country, role } = req.body; const hash = await bcrypt.hash(password, 12); try { db.prepare("INSERT INTO users VALUES (?,?,?,?,?,?,?,?)") .run(id(), name, email.toLowerCase(), hash, country, role, role === "patient" ? "approved" : "pending", now()); res.status(201).json({ message: "Registered" }); } catch { res.status(400).json({ error: "Registration failed" }); } }); app.post("/api/auth/login", async (req, res) => { const { email, password } = req.body; const user = db.prepare("SELECT * FROM users WHERE email=?").get(email.toLowerCase()); if (user && await bcrypt.compare(password, user.password_hash)) { const token = jwt.sign({ sub: user.id }, JWT_SECRET, { expiresIn: "8h" }); res.cookie("radcare_token", token, { httpOnly: true, secure: process.env.NODE_ENV === "production" }); res.json({ user: publicUser(user) }); } else res.status(401).json({ error: "Invalid credentials" }); }); // Profile Update Route (Fixed) app.put("/api/patient/profile", auth, (req, res) => { const { phone, dateOfBirth, emergencyContact } = req.body; const existing = db.prepare("SELECT id FROM patient_profiles WHERE user_id=?").get(req.user.id); if (existing) { db.prepare("UPDATE patient_profiles SET phone=?, date_of_birth=?, emergency_contact=? WHERE user_id=?") .run(phone, dateOfBirth, emergencyContact, req.user.id); } else { db.prepare("INSERT INTO patient_profiles VALUES (?,?,?,?,?,?)") .run(id(), req.user.id, phone, dateOfBirth, emergencyContact, now()); } res.json({ ok: true }); }); app.listen(PORT, () => console.log(`Server running on http://localhost:${PORT}`));
